cmake_minimum_required(VERSION 3.18)
project(original_model_mpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(original_model_bridge SHARED bridge.cpp generated/ilqr.cpp generated/config_loader.cpp generated/support.cpp)
# Resolve Eigen from the project root instead of a directory-depth assumption.
set(EXPERIMENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
while(NOT EXISTS "${EXPERIMENT_ROOT}/AGENTS.md")
  get_filename_component(EXPERIMENT_PARENT "${EXPERIMENT_ROOT}" DIRECTORY)
  if(EXPERIMENT_PARENT STREQUAL EXPERIMENT_ROOT)
    message(FATAL_ERROR "Cannot locate project root containing AGENTS.md")
  endif()
  set(EXPERIMENT_ROOT "${EXPERIMENT_PARENT}")
endwhile()
set(EXPERIMENT_EIGEN "${EXPERIMENT_ROOT}/A02_对照组代码/公共依赖/eigen3")
if(NOT EXISTS "${EXPERIMENT_EIGEN}/Eigen/Core")
  set(EXPERIMENT_EIGEN "${EXPERIMENT_ROOT}/01_修改后代码/third_party/eigen3")
endif()
if(NOT EXISTS "${EXPERIMENT_EIGEN}/Eigen/Core")
  message(FATAL_ERROR "Retained Eigen dependency was not found")
endif()
target_include_directories(original_model_bridge PRIVATE generated/include "${EXPERIMENT_EIGEN}")
target_compile_definitions(original_model_bridge PRIVATE NOMINMAX _USE_MATH_DEFINES)
if(MSVC)
  target_compile_options(original_model_bridge PRIVATE /utf-8)
else()
  target_compile_options(original_model_bridge PRIVATE -Wa,-mbig-obj)
endif()
set_target_properties(original_model_bridge PROPERTIES PREFIX "")
