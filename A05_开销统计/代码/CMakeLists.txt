cmake_minimum_required(VERSION 3.18)
project(original_model_mpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(original_model_bridge SHARED bridge.cpp generated/ilqr.cpp generated/config_loader.cpp generated/support.cpp)
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Directory containing Eigen headers")
if(NOT EIGEN_INCLUDE_DIR)
  get_filename_component(PROJECT_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  if(EXISTS "${PROJECT_WORKSPACE}/A02_对照组代码/公共依赖/eigen3/Eigen/Core")
    set(EIGEN_INCLUDE_DIR "${PROJECT_WORKSPACE}/A02_对照组代码/公共依赖/eigen3")
  elseif(EXISTS "${PROJECT_WORKSPACE}/01_修改后代码/third_party/eigen3/Eigen/Core")
    set(EIGEN_INCLUDE_DIR "${PROJECT_WORKSPACE}/01_修改后代码/third_party/eigen3")
  else()
    message(FATAL_ERROR "Set EIGEN_INCLUDE_DIR to the project Eigen headers")
  endif()
endif()
target_include_directories(original_model_bridge PRIVATE generated/include "${EIGEN_INCLUDE_DIR}")
target_compile_definitions(original_model_bridge PRIVATE NOMINMAX _USE_MATH_DEFINES)
if(MSVC)
  target_compile_options(original_model_bridge PRIVATE /utf-8)
else()
  target_compile_options(original_model_bridge PRIVATE -Wa,-mbig-obj)
endif()
set_target_properties(original_model_bridge PROPERTIES PREFIX "")
